```python
import pytest

def test_merge_intervals_empty_input():
    assert merge_intervals([]) == []

def test_merge_intervals_single_interval():
    assert merge_intervals([(1, 2)]) == [(1, 2)]

def test_merge_intervals_no_overlap():
    assert merge_intervals([(1, 2), (3, 4), (5, 6)]) == [(1, 2), (3, 4), (5, 6)]

def test_merge_intervals_full_overlap():
    assert merge_intervals([(1, 5), (2, 3), (4, 7)]) == [(1, 7)]

def test_merge_intervals_partial_overlap():
    assert merge_intervals([(1, 3), (2, 5), (6, 8)]) == [(1, 5), (6, 8)]

def test_merge_intervals_multiple_overlaps():
    assert merge_intervals([(1, 10), (2, 5), (6, 8), (11, 15)]) == [(1, 10), (11, 15)]

def test_merge_intervals_with_same_start():
    assert merge_intervals([(1, 3), (1, 4), (1, 5)]) == [(1, 5)]

def test_merge_intervals_with_same_end():
    assert merge_intervals([(1, 5), (2, 5), (3, 5)]) == [(1, 5)]

def test_merge_intervals_with_negative_numbers():
    assert merge_intervals([(-5, -2), (-3, 0), (-1, 1)]) == [(-5, 1)]

def test_merge_intervals_with_one_interval_and_one_overlapping():
    assert merge_intervals([(1, 5), (3, 7)]) == [(1, 7)]

def test_merge_intervals_with_one_interval_and_one_non_overlapping():
    assert merge_intervals([(1, 5), (6, 8)]) == [(1, 5), (6, 8)]
```
