```python
import pytest
from t3_pure import merge_intervals


class TestMergeIntervals:
    def test_empty_input(self):
        assert merge_intervals([]) == []

    def test_single_interval(self):
        assert merge_intervals([(1, 2)]) == [(1, 2)]

    def test_non_overlapping_intervals(self):
        assert merge_intervals([(1, 2), (3, 4), (5, 6)]) == [(1, 2), (3, 4), (5, 6)]

    def test_overlapping_intervals(self):
        assert merge_intervals([(1, 3), (2, 4), (5, 7)]) == [(1, 4), (5, 7)]

    def test_adjacent_intervals(self):
        assert merge_intervals([(1, 2), (2, 3)]) == [(1, 3)]

    def test_fully_contained_intervals(self):
        assert merge_intervals([(1, 5), (2, 3), (4, 6)]) == [(1, 6)]

    def test_already_sorted_intervals(self):
        assert merge_intervals([(1, 2), (2, 3), (3, 4)]) == [(1, 4)]

    def test_large_overlap(self):
        assert merge_intervals([(1, 10), (2, 3), (5, 8)]) == [(1, 10)]

    def test_negative_numbers(self):
        assert merge_intervals([(-5, -3), (-4, -2)]) == [(-5, -2)]

    def test_single_point_intervals(self):
        assert merge_intervals([(1, 1), (1, 1), (2, 2)]) == [(1, 1), (2, 2)]

    def test_input_not_mutated(self):
        original = [(1, 2), (2, 3)]
        result = merge_intervals(original)
        assert original == [(1, 2), (2, 3)]
```
