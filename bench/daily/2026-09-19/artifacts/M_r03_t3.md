```python
import pytest
from t3_pure import merge_intervals

def test_merge_intervals_empty():
    assert merge_intervals([]) == []

def test_merge_intervals_single():
    assert merge_intervals([(1, 2)]) == [(1, 2)]

def test_merge_intervals_overlapping():
    assert merge_intervals([(1, 3), (2, 4)]) == [(1, 4)]

def test_merge_intervals_adjacent():
    assert merge_intervals([(1, 2), (2, 3)]) == [(1, 3)]

def test_merge_intervals_disjoint():
    assert merge_intervals([(1, 2), (3, 4)]) == [(1, 2), (3, 4)]

def test_merge_intervals_multiple():
    assert merge_intervals([(1, 5), (2, 3), (4, 7), (6, 8)]) == [(1, 8)]

def test_merge_intervals_with_negative_numbers():
    assert merge_intervals([(-5, -1), (-3, 2)]) == [(-5, 2)]
```
